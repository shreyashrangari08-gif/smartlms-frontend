const express = require('express'); const mongoose = require('mongoose'); const cors = require('cors'); const User = require('./user'); // Aapki file ka name user.js hai (small u) const Course = require('./course'); // ========================================== // 🟢 AUTH ROUTES (BILKUL CHHEDA NAHI HAI) // ========================================== app.post('/register', async (req, res) => { try { const { username, email } = req.body; } }); app.post('/login', async (req, res) => { try { const { username, email } = req.body; const user = await User.findOne({ email }); if (user && user.username === username) { res.status(200).json({ message: 'Login Successful!' }); } else { res.status(401).json({ message: 'Invalid credentials.' }); } } catch (err) { res.status(500).json({ message: 'Login Error: ' + err.message }); } }); // ========================================== // 🟢 COURSE ROUTES (KUCH BHI CHANGE NAHI KIYA) // ========================================== app.post('/add-course', async (req, res) => { try { const newCourse = new Course(req.body); res.status(200).json(courses); } catch (err) { res.status(500).json({ message: 'Error: ' + err.message }); } }); app.get('/course/:id', async (req, res) => { try { const course = await Course.findById(req.params.id);
