Assessment

` + mod.quizzes.map((q, i) => `

${q.q}

`).join('') + ``; } // Final Assessment load karne ke liye naya function function loadFinal() { const final = courseData.finalAssessment; document.getElementById('lessonTitle').innerText = final.title; document.getElementById('videoPlayer').style.display = "none"; // Video chupa do const quizBox = document.querySelector('.quiz-box'); quizBox.innerHTML = `

Final Assessment

` + final.questions.map((q, i) => `

${i+1}. ${q.q}

`).join('') + ``; }
